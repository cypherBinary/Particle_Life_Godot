func _compute_step(dt: float) -> void:
	var params_data := PackedFloat32Array([
		max_speed,
		damping_force,
		particle_attraction_distance,
		particle_repulsion_distance,

		dt,
		particleCount,
		max_force,
		particleTypes
	]).to_byte_array()

	rd.buffer_update(
		params_buffer,
		0,
		params_data.size(),
		params_data
	)

	var compute_list := rd.compute_list_begin()

	rd.compute_list_bind_compute_pipeline(
		compute_list,
		pipeline
	)

	rd.compute_list_bind_uniform_set(
		compute_list,
		uniform_set,
		0
	)

	rd.compute_list_bind_uniform_set(
		compute_list,
		params_uniform_set,
		1
	)

	rd.compute_list_dispatch(
		compute_list,
		particleCount,
		1,
		1
	)

	rd.compute_list_end()

	rd.submit()
	rd.sync()

	# getting output
	var position_data := rd.buffer_get_data(position_buffer)
	var velocity_data := rd.buffer_get_data(velocity_buffer)

	particles = position_data.to_vector2_array()
	velocities = velocity_data.to_vector2_array()

func _draw_multimesh_particles() -> void:
	for i in range(particleCount):
		particle_multimesh.set_instance_transform_2d(
			i,

		)
		particle_multimesh.set_instance_color(
			i,
			types[particle_types[i]]
		)

func _particle_logic(dt: float) -> void:
	grid.rebuild(particles)

	var radius_offset := Vector2.ONE * particle_attraction_distance
	for i in range(particleCount):
		var force := grid.calculate_force(
			particles[i],
			particle_attraction_distance,
			particle_repulsion_distance,
			particles,
			particle_types,
			particle_types[i],
			rules,
			particleTypes,
			max_speed,
			radius_offset
		)

		var desired_vel := force.limit_length(max_speed)

		var vel := velocities[i]
		vel = vel.lerp(desired_vel, damping_force)
		vel = vel.limit_length(max_speed)

		particles[i] += vel * dt
		velocities[i] = vel

func _logic_setup():
	rnd_generator.seed = 168
	
	particleCount = particleCountPerType * particleTypes

	# Rules: randomize only when requested or when the matrix size changes.
	if randomize_rules_on_reset or rules.size() != particleTypes * particleTypes:
		rules.resize(particleTypes * particleTypes)

		for i in range(rules.size()):
			rules[i] = (
				roundf(Utils.Rnd.randf() * attractionDetail)
				/ attractionDetail
			) * 2.0 - 1.0

	# Particles: regenerate positions and velocities every reset.
	particles.resize(particleCount)
	velocities.resize(particleCount)
	particle_types.resize(particleCount)

	for i in range(particleCountPerType):
		var val := particleTypes * i

		for j in range(particleTypes):
			var index := val + j

			particles[index] = Vector2(
				Utils.Rnd.range_f(-world_size.x, world_size.x),
				Utils.Rnd.range_f(-world_size.y, world_size.y)
			)

			particle_types[index] = j

			velocities[index] = Vector2(
				Utils.Rnd.range_f(-max_speed, max_speed),
				Utils.Rnd.range_f(-max_speed, max_speed)
			)

func _particle_rendering_setup():
	particle_mesh.size = Vector2(particlesSize * 2, particlesSize * 2)

	var multi_shader := Shader.new()
	multi_shader.code = """
	shader_type canvas_item;

	void fragment() {
		vec2 p = UV - vec2(0.5);

		float d = length(p) - 0.5;
		float aa = fwidth(d);

		COLOR.a *= 1.0 - smoothstep(-aa, aa, d);
	}
	"""

	particle_material.shader = multi_shader
	particle_renderer.material = particle_material

	particle_multimesh.mesh = particle_mesh
	particle_multimesh.transform_format = MultiMesh.TRANSFORM_2D
	particle_multimesh.use_colors = true
	particle_multimesh.instance_count = particleCount

	particle_renderer.multimesh = particle_multimesh

	add_child(particle_renderer)

@export var move_speed := 1000.0
@export var shift_multiplier := 3.0
@export var zoom_speed := 1.15
@export var min_zoom := 0.1
@export var max_zoom := 5.0

func _process(dt: float) -> void:
	var direction := Vector2(
		(Input.is_key_pressed(KEY_D) as float) - (Input.is_key_pressed(KEY_A) as float),
		(Input.is_key_pressed(KEY_S) as float) - (Input.is_key_pressed(KEY_W) as float)
	)

	if direction.length_squared() > 0.0:
		direction = direction.normalized()

	var speed := move_speed

	if Input.is_key_pressed(KEY_SHIFT):
		speed *= shift_multiplier

	position += direction * speed * dt

	if Input.is_key_pressed(KEY_E):
		zoom *= pow(zoom_speed, dt * 10.0)

	if Input.is_key_pressed(KEY_Q):
		zoom /= pow(zoom_speed, dt * 10.0)

	zoom.x = clamp(zoom.x, min_zoom, max_zoom)
	zoom.y = clamp(zoom.y, min_zoom, max_zoom)

func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_WHEEL_UP and event.pressed:
			zoom *= zoom_speed

		elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN and event.pressed:
			zoom /= zoom_speed

		zoom.x = clamp(zoom.x, min_zoom, max_zoom)
		zoom.y = clamp(zoom.y, min_zoom, max_zoom)

extends Control
class_name RuleMatrix

signal rule_changed(input_type: int, target_type: int, value: float)

var rules: PackedFloat32Array
var particle_types: int = 3
var type_colors: Dictionary = {}

var cell_size := 35.0
var label_size := 15.0
var cell_gap := 2.0


func setup(
	type_count: int,
	initial_rules: PackedFloat32Array,
	colors: Dictionary
) -> void:
	particle_types = type_count
	rules = initial_rules
	type_colors = colors

	custom_minimum_size = Vector2(
		label_size + particle_types * (cell_size + cell_gap),
		label_size + particle_types * (cell_size + cell_gap)
	)

	queue_redraw()


func _draw() -> void:
	if rules.size() != particle_types * particle_types:
		return

	var font := ThemeDB.fallback_font
	var font_size := 14

	# Column headers: target particle types.
	for target in range(particle_types):
		var x := label_size + target * (cell_size + cell_gap)
		draw_circle(
			Vector2(x + cell_size / 2.0, label_size / 2.0),
			6.0,
			type_colors[target % type_colors.size()],
			true,
			-1,
			true
		)

	# Row headers: input particle types.
	for input_type in range(particle_types):
		var y := label_size + input_type * (cell_size + cell_gap)
		draw_circle(
			Vector2(label_size / 2.0, y + cell_size / 2.0),
			6.0,
			type_colors[input_type % type_colors.size()],
			true,
			-1,
			true
		)

	# Rule cells.
	for input_type in range(particle_types):
		for target in range(particle_types):
			var index := input_type * particle_types + target
			var value := rules[index]

			var x := label_size + target * (cell_size + cell_gap)
			var y := label_size + input_type * (cell_size + cell_gap)
			var rect := Rect2(x, y, cell_size, cell_size)

			draw_rect(rect, _value_color(value), true)
			draw_rect(rect, Color(0.35, 0.35, 0.35), false, 1.0)

			var text := "%.1f" % value
			var text_width := font.get_string_size(
				text, HORIZONTAL_ALIGNMENT_LEFT, -1, font_size
			).x

			draw_string(
				font,
				Vector2(
					x + (cell_size - text_width) / 2.0,
					y + cell_size / 2.0 + font_size * 0.35
				),
				text,
				HORIZONTAL_ALIGNMENT_LEFT,
				-1,
				font_size,
				Color.WHITE
			)


func _value_color(value: float) -> Color:
	if value < 0.0:
		return Color(0.95, 0.12, 0.16).lerp(
			Color(0.10, 0.10, 0.10), clampf(value + 1.0, 0.0, 1.0)
		)

	return Color(0.10, 0.10, 0.10).lerp(
		Color(0.10, 0.85, 0.35), clampf(value, 0.0, 1.0)
	)


func _gui_input(event: InputEvent) -> void:
	if not (event is InputEventMouseButton):
		return
	if not event.pressed:
		return

	var mouse: Vector2 = event.position
	var column := int((mouse.x - label_size) / (cell_size + cell_gap))
	var row := int((mouse.y - label_size) / (cell_size + cell_gap))

	if row < 0 or row >= particle_types:
		return
	if column < 0 or column >= particle_types:
		return

	var local_x := fposmod(mouse.x - label_size, cell_size + cell_gap)
	var local_y := fposmod(mouse.y - label_size, cell_size + cell_gap)

	# Ignore clicks in the gaps between cells.
	if local_x >= cell_size or local_y >= cell_size:
		return

	var index := row * particle_types + column

	if event.button_index == MOUSE_BUTTON_LEFT:
		rules[index] = minf(2.0, snappedf(rules[index] + 0.1, 0.1))
	elif event.button_index == MOUSE_BUTTON_RIGHT:
		rules[index] = maxf(-1.0, snappedf(rules[index] - 0.1, 0.1))
	else:
		return

	rule_changed.emit(row, column, rules[index])
	queue_redraw()

func refresh(new_rules: PackedFloat32Array) -> void:
	rules = new_rules
	queue_redraw()

#[compute]
#version 450

layout(local_size_x = 1) in;

layout(set = 0, binding = 0, std430) buffer Positions {
    vec2 positions[];
};

layout(set = 0, binding = 1, std430) buffer Velocities {
    vec2 velocities[];
};

layout(set = 0, binding = 2, std430) buffer Types {
    int types[];
};

layout(set = 0, binding = 3, std430) buffer Rules {
    float rules[];
};

layout(set = 1, binding = 0, std140) uniform Params {
    vec4 params_a;
    vec4 params_b;
};

void main() {
    uint id = gl_GlobalInvocationID.x;

    vec2 center = positions[id];
    int input_type = types[id];

    vec2 force = vec2(0.0);

    float repulsion_distance = params_a.w;
    float attraction_distance = params_a.z;
    float max_speed = params_a.x;

    float interaction_distance = repulsion_distance + attraction_distance;

    float repulsion_squared = repulsion_distance * repulsion_distance;
    float interaction_squared = interaction_distance * interaction_distance;

    float attraction_half = attraction_distance / 2.0;

    int rule_base = input_type * int(params_b.w);

    for (uint i = 0; i < uint(params_b.y); i++) {
        vec2 diff = positions[i] - center;
        float distance_squared = dot(diff, diff);

        if (distance_squared == 0.0) {
            continue;
        }

        if (distance_squared <= interaction_squared) {
            float distance = sqrt(distance_squared);

            if (distance <= repulsion_distance) {
                // -1 at 0, 0 at repulsion distance
                float force_value =
                    -1.0 + distance / repulsion_distance;

                // Negative force along diff = away from neighbor
                force += diff / distance
                    * (max_speed * force_value);
            }
            else {
                // 0 -> 1 -> 0 across attraction range
                float d = distance - repulsion_distance;
                float t;

                if (d <= attraction_half) {
                    t = d / attraction_half;
                }
                else {
                    t = 1.0 - (d - attraction_half) / attraction_half;
                }

                t *= params_b.z;

                int rule_index = rule_base + types[i];

                force += diff / distance
                    * (max_speed * t)
                    * rules[rule_index];
            }
        }
    }

    vec2 desired_velocity = force;

    float force_length = length(desired_velocity);

    if (force_length > max_speed) {
        desired_velocity =
            desired_velocity / force_length * max_speed;
    }

    vec2 velocity = velocities[id];

    velocity = mix(
        velocity,
        desired_velocity,
        params_a.y
    );

    float velocity_length = length(velocity);

    if (velocity_length > max_speed) {
        velocity =
            velocity / velocity_length * max_speed;
    }

    positions[id] += velocity * params_b.x;
    velocities[id] = velocity;
}

class Grid2D:
	var cell_size: float
	var cells: Dictionary[Vector2i, int] = {}
	var next: PackedInt32Array
	
	var checked_particles := 0
	var found_particles := 0
	
	func _init(
		p_cell_size: float,
		item_count: int
	) -> void:
		cell_size = p_cell_size
		next.resize(item_count)
	
	func clear() -> void:
		cells.clear()
	
	func rebuild(positions: PackedVector2Array) -> void:
		clear()
		
		for i in positions.size():
			var cell := world_to_cell(positions[i])
			
			var previous: int = cells.get(cell, -1)

			next[i] = previous
			cells[cell] = i
	
	func world_to_cell(position: Vector2) -> Vector2i:
		return Vector2i(
			floori(position.x / cell_size),
			floori(position.y / cell_size)
		)
	
	func circle_query(
		center: Vector2,
		radius: float,
		positions: PackedVector2Array,
		callback: Callable
	) -> Vector2i:
		checked_particles = 0
		found_particles = 0
		
		var min_cell := world_to_cell(center - Vector2.ONE * radius)
		var max_cell := world_to_cell(center + Vector2.ONE * radius)
		
		var radius_squared := radius * radius
		
		for y in range(min_cell.y, max_cell.y + 1):
			for x in range(min_cell.x, max_cell.x + 1):
				var cell := Vector2i(x, y)
				
				if not cells.has(cell):
					continue
				
				var particle := cells[cell]
				
				while particle != -1:
					var offset := positions[particle] - center
					
					checked_particles += 1
					
					if offset.length_squared() <= radius_squared:
						found_particles += 1
						callback.call(particle, offset)
					
					particle = next[particle]
		
		return Vector2i(checked_particles, found_particles)
		
	func calculate_force(
		center: Vector2,
		radius: float,
		repulsion_distance: float,
		positions: PackedVector2Array,
		particle_types: PackedInt32Array,
		input_type: int,
		rules: PackedFloat32Array,
		type_count: int,
		max_speed: float,
		vector_radius: Vector2
	) -> Vector2:
		var force := Vector2.ZERO

		var min_cell := world_to_cell(center - vector_radius)
		var max_cell := world_to_cell(center + vector_radius)

		var radius_squared := radius * radius
		var repulsion_squared := repulsion_distance * repulsion_distance
		var half := radius / 2.0
		var rule_base := input_type * type_count

		for y in range(min_cell.y, max_cell.y + 1):
			for x in range(min_cell.x, max_cell.x + 1):
				var cell := Vector2i(x, y)

				var particle: int = cells.get(cell, -1)

				if particle == -1:
					continue

				while particle != -1:
					var diff := positions[particle] - center
					var distance_squared := diff.length_squared()

					if distance_squared <= radius_squared:
						var distance := sqrt(distance_squared)
						
						if distance_squared == 0.0:
							particle = next[particle]
							continue

						if distance_squared <= repulsion_squared:
							force -= diff / distance * remap(
								distance,
								repulsion_distance,
								0.0,
								0.0,
								max_speed
							)
						else:
							var t: float = 1.0 - (abs(distance - half) / half)

							var rule_index := rule_base + particle_types[particle]

							force += diff / distance \
								* lerpf(0.0, max_speed, t) \
								* rules[rule_index]

					particle = next[particle]

		return force
